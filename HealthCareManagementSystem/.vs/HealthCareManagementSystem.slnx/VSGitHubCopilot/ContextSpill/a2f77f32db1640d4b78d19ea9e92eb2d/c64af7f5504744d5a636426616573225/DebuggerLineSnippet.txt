61:         {

63:             {

65:                 return View(doctorVM);
66:             }

70:             if (!await _clinicRepository.AnyAsync(c => c.Id == clinicId))
71:             {
72:                 ModelState.AddModelError(nameof(DoctorVM.ClinicId), "The selected clinic does not exist.");
73:                 await PopulateClinicsAsync();
74:                 return View(doctorVM);
75:             }

77:             string? imageName = null;
78: 
79:             if (doctorVM.Image is { Length: > 0 })

82: 
83:                 if (!saved.Succeeded)
84:                 {
85:                     ModelState.AddModelError(nameof(DoctorVM.Image), saved.Error ?? "The image could not be saved.");
86:                     await PopulateClinicsAsync(clinicId);
87:                     return View(doctorVM);
88:                 }
89: 
90:                 imageName = saved.FileName;
91:             }
92: 
93:             var doctor = new Doctor
94:             {
95:                 Name = doctorVM.Name.Trim(),
96:                 Specialization = doctorVM.Specialization.Trim(),
97:                 ClinicId = clinicId,
98:                 ImageName = imageName
99:             };
100: 
101:             try
102:             {
103:                 await _doctorRepository.AddAsync(doctor);
104:                 await _doctorRepository.CommitAsync();
105:             }
106:             catch (DbUpdateException)
107:             {
108:                 _imageService.Delete(imageName, ImageFolder);
109:                 ModelState.AddModelError(string.Empty, "The doctor could not be saved. Please try again.");
110:                 await PopulateClinicsAsync(clinicId);
111:                 return View(doctorVM);
112:             }
113: 
114:             TempData["Success-Notification"] = "Doctor added successfully.";
115: 
116:             return RedirectToAction(nameof(Index));
117:         }
